> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build an optional `comix-dl tui` full-screen Textual interface that drives the existing application layer directly instead of wrapping prompt-based CLI flows.

**Architecture:** Add a new `comix_dl.core.tui` package with pure state reducers, an application controller, and Textual screens. Keep the current CLI as the scriptable interface; the TUI uses `ApplicationSession`, repositories, cleanup use cases, and download events directly. Use Textual workers for search, series loading, and downloads so the UI remains responsive.

**Tech Stack:** Python 3.11+, Textual 8.x, Rich renderables inside Textual widgets, pytest, pytest-asyncio, mypy, ruff.

def run_tui(*, mirror: str | None = None) -> int:

    return int(result or 0)

```python
"""Pure state helpers for the Textual TUI."""

from __future__ import annotations

import re
from dataclasses import dataclass, replace
from typing import Literal

from comix_dl.core.application.download_usecase import DownloadChapterEvent, DownloadSummary
from comix_dl.core.cli.display import format_bytes
from comix_dl.core.models import ChapterInfo


OutputFormat = Literal["pdf", "cbz", "both"]
DownloadStatus = Literal[
    "queued",
    "skipped",
    "started",
    "planned",
    "progress",
    "missing_images",
    "failed",
    "partial",
    "converted",
    "conversion_failed",
]


@dataclass(frozen=True)
class DownloadRequest:
    """Immutable input for one TUI download batch."""

    series_title: str
    chapters: tuple[ChapterInfo, ...]
    fmt: OutputFormat
    optimize: bool


@dataclass
class ChapterSelectionState:
    """Filter and selection state for a chapter list."""

    chapters: list[ChapterInfo]
    visible_chapters: list[ChapterInfo]
    selected_ids: set[int]
    status: str = ""

    @classmethod
    def from_chapters(cls, chapters: list[ChapterInfo]) -> ChapterSelectionState:
        return cls(chapters=list(chapters), visible_chapters=list(chapters), selected_ids=set())

    @property
    def selected_count(self) -> int:
        return len(self.selected_ids)

    @property
    def selected_chapters(self) -> list[ChapterInfo]:
        return [chapter for chapter in self.chapters if chapter.chapter_id in self.selected_ids]

    def apply_filter(self, raw_filter: str) -> None:
        raw = raw_filter.strip()
        if not raw:
            self.visible_chapters = list(self.chapters)
            self.status = f"Showing all {len(self.visible_chapters)} chapter(s)."
            return

        keep_words: list[str] = []
        remove_words: list[str] = []
        for token in re.findall(r"[+\-]?\S+", raw):
            if token.startswith("+"):
                word = token[1:].strip().lower()
                if word:
                    keep_words.append(word)
            elif token.startswith("-"):
                word = token[1:].strip().lower()
                if word:
                    remove_words.append(word)
            else:
                keep_words.append(token.lower())

        filtered = list(self.chapters)
        if keep_words:
            filtered = [
                chapter for chapter in filtered
                if any(word in chapter.title.lower() for word in keep_words)
            ]
        if remove_words:
            filtered = [
                chapter for chapter in filtered
                if not any(word in chapter.title.lower() for word in remove_words)
            ]

        if not filtered:
            self.status = f"No chapters matched '{raw}'; keeping previous filter."
            return

        self.visible_chapters = filtered
        self.status = f"{len(filtered)} chapter(s) visible."

    def toggle(self, chapter_id: int) -> None:
        if chapter_id in self.selected_ids:
            self.selected_ids.remove(chapter_id)
            return
        self.selected_ids.add(chapter_id)

    def select_visible(self) -> None:
        self.selected_ids.update(chapter.chapter_id for chapter in self.visible_chapters)
        self.status = f"Selected {len(self.visible_chapters)} visible chapter(s)."

    def clear_visible(self) -> None:
        for chapter in self.visible_chapters:
            self.selected_ids.discard(chapter.chapter_id)
        self.status = f"Cleared {len(self.visible_chapters)} visible chapter(s)."

    def clear_all(self) -> None:
        self.selected_ids.clear()
        self.status = "Selection cleared."


@dataclass(frozen=True)
class DownloadRow:
    """One row in the TUI download table."""

    chapter_id: int
    title: str
    status: DownloadStatus = "queued"
    completed: int = 0
    total: int | None = None
    detail: str = ""

    @property
    def progress_text(self) -> str:
        if self.total is None:
            return "..."
        if self.total <= 0:
            return "0/0"
        return f"{self.completed}/{self.total}"


@dataclass
class DownloadRowsState:
    """Reducer for download events emitted by the application layer."""

    rows: dict[int, DownloadRow]

    @classmethod
    def from_chapters(cls, chapters: list[ChapterInfo] | tuple[ChapterInfo, ...]) -> DownloadRowsState:
        return cls(rows={
            chapter.chapter_id: DownloadRow(chapter_id=chapter.chapter_id, title=chapter.title)
            for chapter in chapters
        })

    def apply(self, event: DownloadChapterEvent) -> DownloadRow:
        current = self.rows.get(
            event.chapter_id,
            DownloadRow(chapter_id=event.chapter_id, title=event.chapter_title),
        )

        if event.kind == "skipped":
            updated = replace(current, status="skipped", completed=1, total=1, detail="Already complete")
        elif event.kind == "started":
            updated = replace(current, status="started", detail="Starting")
        elif event.kind == "planned":
            updated = replace(current, status="planned", completed=0, total=event.total, detail="Planned")
        elif event.kind == "progress":
            total = event.total if event.total is not None else current.total
            updated = replace(current, status="progress", completed=event.completed, total=total, detail="Downloading")
        elif event.kind == "missing_images":
            updated = replace(current, status="missing_images", detail="No images from remote API")
        elif event.kind == "failed":
            updated = replace(current, status="failed", detail=event.message or "Download failed")
        elif event.kind == "partial":
            updated = replace(current, status="partial", detail=event.message or "Incomplete chapter")
        elif event.kind == "converted":
            total = current.total if current.total is not None else event.total
            completed = total if total is not None else current.completed
            updated = replace(
                current,
                status="converted",
                completed=completed,
                total=total,
                detail=event.output_name or event.chapter_title,
            )
        else:
            updated = replace(
                current,
                status="conversion_failed",
                detail=event.message or "Conversion failed",
            )

        self.rows[event.chapter_id] = updated
        return updated


def format_summary_line(summary: DownloadSummary) -> str:
    """Return a compact one-line batch summary for the TUI."""
    parts: list[str] = []
    if summary.completed:
        parts.append(f"{summary.completed} completed")
    if summary.skipped:
        parts.append(f"{summary.skipped} skipped")
    if summary.partial:
        parts.append(f"{summary.partial} partial")
    if summary.failed:
        parts.append(f"{summary.failed} failed")
    if not parts:
        parts.append("No chapters completed")

    speed = summary.total_bytes / summary.elapsed_seconds if summary.elapsed_seconds > 0 else 0
    parts.append(format_bytes(summary.total_bytes))
    parts.append(f"{format_bytes(int(speed))}/s")
    parts.append(f"{summary.elapsed_seconds:.1f}s")
    return " · ".join(parts)
```

"""Application controller for the full-screen TUI."""

from __future__ import annotations

from collections.abc import AsyncIterator, Callable
from contextlib import AbstractAsyncContextManager
from pathlib import Path
from typing import Protocol

from comix_dl.core.application.cleanup_usecase import (
    CleanupPlan,
    CleanupResult,
    DownloadedSeries,
    apply_cleanup_plan,
    build_cleanup_plan,
    list_downloaded_series,
)
from comix_dl.core.application.download_usecase import DownloadChapterEvent, DownloadEventHandler, DownloadSummary
from comix_dl.core.application.session import ApplicationSession, open_application_session
from comix_dl.core.history import HistoryEntry, HistoryRepository
from comix_dl.core.models import SearchResult, SeriesInfo
from comix_dl.core.settings import Settings, SettingsRepository
from comix_dl.core.tui.state import DownloadRequest


class SessionFactory(Protocol):
    def __call__(self, **kwargs: object) -> AbstractAsyncContextManager[ApplicationSession]:
        """Create an async context manager yielding an application session."""


class TuiController:
    """Textual-independent coordinator for application use cases."""

    def __init__(
        self,
        *,
        session_factory: SessionFactory = open_application_session,
        settings_repository: SettingsRepository | None = None,
        history_repository: HistoryRepository | None = None,
        mirror: str | None = None,
    ) -> None:
        self._session_factory = session_factory
        self._settings_repository = settings_repository or SettingsRepository()
        self._history_repository = history_repository or HistoryRepository()
        self._mirror = mirror
        self._context: AbstractAsyncContextManager[ApplicationSession] | None = None
        self._session: ApplicationSession | None = None
        self._shutdown_requested = False

    @property
    def is_open(self) -> bool:
        return self._session is not None

    @property
    def settings(self) -> Settings:
        if self._session is not None:
            return self._session.settings
        return self._settings_repository.load()

    @property
    def output_dir(self) -> Path:
        if self._session is not None:
            return self._session.output_dir
        return Path(self.settings.output_dir)

    async def open(self) -> None:
        if self._session is not None:
            return
        self._shutdown_requested = False
        self._context = self._session_factory(mirror_override=self._mirror)
        self._session = await self._context.__aenter__()

    async def close(self) -> None:
        context = self._context
        self._context = None
        self._session = None
        if context is not None:
            await context.__aexit__(None, None, None)

    def _require_session(self) -> ApplicationSession:
        if self._session is None:
            raise RuntimeError("TUI session is not open")
        return self._session

    async def search(self, query: str) -> list[SearchResult]:
        return await self._require_session().search(query)

    async def load_series(self, identifier: str) -> SeriesInfo:
        return await self._require_session().load_series(identifier)

    async def download(
        self,
        request: DownloadRequest,
        *,
        on_event: DownloadEventHandler | None = None,
    ) -> DownloadSummary:
        session = self._require_session()
        return await session.download(
            series_title=request.series_title,

class SearchScreen(Widget):

    def __init__(self, controller: TuiController) -> None:
        super().__init__()
        self.controller = controller
        self.results: list[SearchResult] = []

    def compose(self) -> ComposeResult:
        with Vertical():
            yield Static("Search manga", classes="pane-title")
            yield Input(placeholder="Type a manga name and press Enter", id="search-input")
            yield Static("Ready.", id="search-status", classes="muted")
            table = DataTable(id="results")
            table.cursor_type = "row"
            yield table

    def on_mount(self) -> None:
        table = self.query_one("#results", DataTable)
        table.add_columns("#", "Title", "Slug")
        self.query_one("#search-input", Input).focus()

    @on(Input.Submitted, "#search-input")
    async def _submit_search(self, event: Input.Submitted) -> None:
        query = event.value.strip()
        if not query:
            self.query_one("#search-status", Static).update("Enter a search query.")
            return
        self.run_worker(self._search(query), name="search", group="search", exclusive=True, exit_on_error=False)

    async def _search(self, query: str) -> None:
        status = self.query_one("#search-status", Static)
        table = self.query_one("#results", DataTable)
        status.update(f"Searching '{query}'...")
        table.clear()
        self.results = []
        try:
            self.results = await self.controller.search(query)
        except Exception as exc:
            status.update(f"Search failed: {exc}")
            return
        if not self.results:
            status.update("No results found.")
            return
        for index, result in enumerate(self.results, 1):
            table.add_row(str(index), result.title, result.slug, key=str(index - 1))
        status.update(f"{len(self.results)} result(s). Press Enter to open the selected row.")
        table.focus()

    @on(DataTable.RowSelected, "#results")
    async def _open_selected_result(self, event: DataTable.RowSelected) -> None:
        try:
            index = int(str(event.row_key.value))
            result = self.results[index]
        except (ValueError, IndexError):
            self.query_one("#search-status", Static).update("Invalid result selection.")
            return
        self.run_worker(self._load_series(result), name="load-series", group="load-series", exclusive=True, exit_on_error=False)

"""Download progress pane for the TUI."""

from __future__ import annotations

from textual.app import ComposeResult
from textual.containers import Horizontal, Vertical
from textual.widget import Widget
from textual.widgets import Button, DataTable, Static

from comix_dl.core.application.download_usecase import DownloadChapterEvent
from comix_dl.core.cli.display import format_bytes
from comix_dl.core.tui.controller import TuiController
from comix_dl.core.tui.state import DownloadRequest, DownloadRowsState, format_summary_line


class DownloadPane(Widget):
    """Live download progress pane."""

    BINDINGS = [
        ("c", "cancel", "Cancel"),
        ("b", "back_to_search", "Search"),
    ]

    def __init__(self, controller: TuiController, request: DownloadRequest) -> None:
        super().__init__()
        self.controller = controller
        self.request = request
        self.rows = DownloadRowsState.from_chapters(request.chapters)

    def compose(self) -> ComposeResult:
        with Vertical():
            yield Static(f"Downloading {self.request.series_title}", id="download-title", classes="pane-title")
            yield Static("Preparing download...", id="download-status", classes="muted")
            with Horizontal(id="download-actions"):
                yield Button("Cancel", id="cancel-download", variant="warning")
                yield Button("Clean raw images", id="cleanup-button", variant="success", disabled=True)
            table = DataTable(id="download-table")
            table.cursor_type = "row"
            yield table

    def on_mount(self) -> None:
        table = self.query_one("#download-table", DataTable)
        table.add_columns("Chapter", "Status", "Progress", "Detail")
        self._refresh_table()
        self.run_worker(self._run_download(), name="download", group="download", exit_on_error=False)

    def _refresh_table(self) -> None:
        table = self.query_one("#download-table", DataTable)
        table.clear()
        for row in self.rows.rows.values():
            table.add_row(row.title, row.status, row.progress_text, row.detail, key=str(row.chapter_id))

    def _handle_event(self, event: DownloadChapterEvent) -> None:
        self.rows.apply(event)
        self._refresh_table()

    async def _run_download(self) -> None:
        status = self.query_one("#download-status", Static)
        status.update(f"Downloading {len(self.request.chapters)} chapter(s) as {self.request.fmt.upper()}...")
        try:
            summary = await self.controller.download(self.request, on_event=self._handle_event)
        except Exception as exc:
            status.update(f"Download failed: {exc}")
            return
        status.update(format_summary_line(summary))
        plan = self.controller.cleanup_plan(series_title=self.request.series_title)
        cleanup_button = self.query_one("#cleanup-button", Button)
        cleanup_button.disabled = len(plan.candidates) == 0
        if plan.candidates:
            cleanup_button.label = f"Clean raw images ({format_bytes(plan.total_size_bytes)})"

    def action_cancel(self) -> None:
        self.controller.request_shutdown()
        self.query_one("#download-status", Static).update("Cancel requested. Finishing active work...")

    async def action_back_to_search(self) -> None:
        await self.app.action_show_search()

    def on_button_pressed(self, event: Button.Pressed) -> None:
        if event.button.id == "cancel-download":
            self.action_cancel()
        if event.button.id == "cleanup-button":
            plan = self.controller.cleanup_plan(series_title=self.request.series_title)
            result = self.controller.apply_cleanup(plan)
            self.query_one("#download-status", Static).update(
                f"Cleaned {result.removed_count} directorie(s); {len(result.failed)} failed."
            )
